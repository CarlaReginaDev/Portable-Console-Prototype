# DualShock 2 Buttons mapped to RetroArch default keyboard controls

# --- BUTTON MAPPINGS (Index=Key) ---
0=KEY_X          # Cross (A button) -> X
1=KEY_C          # Circle (B button) -> C
2=KEY_S          # Square (Y button) -> S
3=KEY_D          # Triangle (X button) -> D
4=KEY_Q          # L1 -> Q
5=KEY_W          # R1 -> W
6=KEY_A          # L2 -> A
7=KEY_Z          # R2 -> Z
8=KEY_LEFTSHIFT  # Select -> Left Shift
9=KEY_ENTER      # Start -> Enter
10=KEY_1         # L3 (Left Stick Click) -> 1
11=KEY_2         # R3 (Right Stick Click) -> 2

# --- AXIS MAPPINGS (Index_Direction=Key) ---
# Direction: -1 (NEG) is Left/Up; +1 (POS) is Right/Down

# Left Stick (Axis 0 & 1) - Mapped to Arrow Keys
0_-1=KEY_LEFT    # Left Stick Left
0_1=KEY_RIGHT   # Left Stick Right
1_-1=KEY_UP     # Left Stick Up
1_1=KEY_DOWN    # Left Stick Down

# Right Stick (Axis 2 & 3) - Mapped to WASD
2_-1=KEY_A      # Right Stick Left
2_1=KEY_D       # Right Stick Right
3_-1=KEY_W      # Right Stick Up
3_1=KEY_S       # Right Stick Down

# D-Pad (Axis 6 & 7) - Mapped to Arrow Keys 
# (These often handle the D-Pad on modern controllers)
6_-1=KEY_LEFT    # D-Pad Left
6_1=KEY_RIGHT   # D-Pad Right
7_-1=KEY_UP     # D-Pad Up
7_1=KEY_DOWN    # D-Pad Down
